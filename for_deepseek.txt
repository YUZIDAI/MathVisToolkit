# 🌐 MathVisLab — 微积分几何可视化平台

> 工科数学分析课程教学团队 · 2026

## 📖 项目简介

MathVisLab 是一个面向微积分教学的**交互式几何可视化平台**，将抽象的数学概念转化为可拖拽、可探索的动态图形。

### 四大核心工具

| 工具 | 入口 | 功能 |
|:---|:---|:---|
| 📐 **切线斜率探索器** | `pages/tangent.html` | 拖拽红点沿曲线移动，实时观察切线与导数值 |
| ∫ **定积分可视化工具** | `pages/integral.html` | 调节矩形数量（1～100），切换采样模式，观察黎曼和如何收敛 |
| ∞ **泰勒逼近演示器** | `pages/taylor.html` | 可调阶数（0～15），多项式逐阶逼近原函数，并显示误差热力图 |
| ◎ **极坐标函数绘制器** | `pages/polar.html` | 彩虹曲线绘制玫瑰线、心形线、螺线，编织动画展示生长过程 |

    function bidirectionalColor(normalizedValue) {

        if (v >= 0) {

    function numericalDerivative(fn, x, h) {
        h = h || 1e-7;
        return (fn(x + h) - fn(x - h)) / (2 * h);

    // ==================== createCoordinateTransformer ====================
    function createCoordinateTransformer(xMin, xMax, yMin, yMax, cW, cH) {
        return {
            toCanvasX: function(mx) { return ((mx - xMin) / (xMax - xMin)) * cW; },
            toCanvasY: function(my) { return cH - ((my - yMin) / (yMax - yMin)) * cH; },
            toMathX: function(cx) { return xMin + (cx / cW) * (xMax - xMin); },
            toMathY: function(cy) { return yMax - (cy / cH) * (yMax - yMin); },
            updateRange: function(a,b,c,d) { xMin=a; xMax=b; yMin=c; yMax=d; },
            updateSize: function(w,h) { cW=w; cH=h; }
        };

    function setupHiDPICanvas(canvas, ctx) {

            canvas.style.height = H + 'px';
            ctx.setTransform(1, 0, 0, 1, 0, 0);

            var dx = points[i+1].x - points[i].x;
            var dy = points[i+1].y - points[i].y;
            len += Math.sqrt(dx*dx + dy*dy);
        }
        return len;
    }

    // ==================== debounce ====================
    function debounce(fn, delay) {
        var timer = null;
        return function() {
            var self = this, args = arguments;
            clearTimeout(timer);
            timer = setTimeout(function() { fn.apply(self, args); }, delay);
        };
    }

    // ==================== getPassedFunction ====================
    function getPassedFunction() {
        var params = new URLSearchParams(window.location.search);
        var func = params.get('func');
        return func ? decodeURIComponent(func) : null;
    }

    // ==================== Storage ====================
    function saveToStorage(key, value) {
        try { localStorage.setItem(key, JSON.stringify(value)); } catch(e) {}
    }
    function loadFromStorage(key, defaultValue) {
        try { var raw = localStorage.getItem(key); return raw ? JSON.parse(raw) : defaultValue; } catch(e) { return defaultValue; }
    }

    // ==================== 导出 ====================
    window.Common = {

        rainbowHSL: rainbowHSL,
        curveLength: curveLength,
        debounce: debounce,
        getPassedFunction: getPassedFunction,
        saveToStorage: saveToStorage,
        loadFromStorage: loadFromStorage
    };

})(window);

        (function() {
            const urlParams = new URLSearchParams(window.location.search);
            const passedFunc = urlParams.get('func');
            if (passedFunc) {
                currentFuncExpr = decodeURIComponent(passedFunc);
                currentFunc = Common.compileExpr(currentFuncExpr);
                // 更新预设按钮激活状态
                document.querySelectorAll('.func-btn').forEach(b => b.classList.remove('active'));
                // 更新自定义输入框

                if (customInput) customInput.value = currentFuncExpr;
                // 延迟重绘

            }

        let a = 0;

        let exactValue = 2;

        // ==================== 自适应视口 ====================
        function autoFitView() {
            const padding = 0.15;
            const xRange = b - a;
            xMin = a - padding * xRange;
            xMax = b + padding * xRange;

            // 采样找y范围
            let ySampledMin = Infinity, ySampledMax = -Infinity;
            const steps = 200;
            for (let i = 0; i <= steps; i++) {
                const x = a + (b - a) * i / steps;
                const y = currentFunc(x);
                if (isFinite(y) && !isNaN(y)) {
                    ySampledMin = Math.min(ySampledMin, y);
                    ySampledMax = Math.max(ySampledMax, y);
                }
            }
            const yRange = Math.max(ySampledMax - ySampledMin, 0.5);
            yMin = Math.min(0, ySampledMin - padding * yRange);
            yMax = Math.max(ySampledMax + padding * yRange, yMin + 1);
        }

        // ==================== Canvas 坐标转换 ====================
        function getGraphRect() {
            return graphCanvas.getBoundingClientRect();
        }

        function toCanvasX(mathX, W) {
            return ((mathX - xMin) / (xMax - xMin)) * W;
        }

        function toCanvasY(mathY, H) {
            return H - ((mathY - yMin) / (yMax - yMin)) * H;
        }

        function toMathX(canvasX, W) {
            return xMin + (canvasX / W) * (xMax - xMin);
        }

        // ==================== 黎曼和计算 ====================
        function computeRiemannSum() {
            const dx = (b - a) / n;
            let sum = 0;
            const errors = [];
            const rectData = [];

            for (let i = 0; i < n; i++) {
                const xLeft = a + i * dx;
                const xRight = a + (i + 1) * dx;
                let sampleX, height;

                switch (mode) {
                    case 'left':
                        sampleX = xLeft;
                        break;
                    case 'right':
                        sampleX = xRight;
                        break;
                    case 'mid':
                        sampleX = (xLeft + xRight) / 2;
                        break;
                    case 'trap':
                        // 梯形：上下底平均
                        const yL = currentFunc(xLeft);
                        const yR = currentFunc(xRight);
                        height = (yL + yR) / 2;
                        sum += height * dx;
                        // 误差取中点
                        const midX = (xLeft + xRight) / 2;
                        const midY = currentFunc(midX);
                        errors.push(Math.abs(midY - height));
                        rectData.push({ xLeft, xRight, yL, yR, height, sampleX: midX, midY });
                        continue;
                    default:
                        sampleX = xLeft;
                }

                if (mode !== 'trap') {
                    height = currentFunc(sampleX);
                    sum += height * dx;
                    const midX = (xLeft + xRight) / 2;
                    const midY = currentFunc(midX);
                    errors.push(Math.abs(midY - height));
                    rectData.push({ xLeft, xRight, height, sampleX, midX, midY });
                }
            }

            return { sum, errors, rectData, dx };
        }

                ctx.lineTo(W, toCanvasY(y, graphH));

                    continue;

                    ctx.lineTo(cxR, cyBottom);
                    ctx.closePath();
                    ctx.fillStyle = fillColor;
                    ctx.fill();
                    ctx.strokeStyle = strokeColor;
                    ctx.lineWidth = 0.8;
                    ctx.stroke();
                } else {
                    // 矩形
                    const cyTop = toCanvasY(rd.height, graphH);
                    const cyBottom = toCanvasY(0, graphH);
                    ctx.fillStyle = fillColor;
                    ctx.fillRect(cxL, Math.min(cyTop, cyBottom), cxR - cxL, Math.abs(cyTop - cyBottom));
                    ctx.strokeStyle = strokeColor;
                    ctx.lineWidth = 0.8;
                    ctx.strokeRect(cxL, Math.min(cyTop, cyBottom), cxR - cxL, Math.abs(cyTop - cyBottom));
                }

                // 误差线段（中点偏差）
                if (mode !== 'trap' && rd.midY !== undefined) {
                    const cyMid = toCanvasY(rd.midY, graphH);
                    const cyH = toCanvasY(rd.height, graphH);
                    ctx.beginPath();
                    ctx.strokeStyle = 'rgba(255,200,100,0.5)';
                    ctx.lineWidth = 1;
                    ctx.setLineDash([2, 2]);
                    const cxMid = (cxL + cxR) / 2;
                    ctx.moveTo(cxMid, Math.min(cyMid, cyH));
                    ctx.lineTo(cxMid, Math.max(cyMid, cyH));
                    ctx.stroke();
                    ctx.setLineDash([]);
                }
            });

            // 积分符号叠加（n≥50时淡入）
            const opacity = n >= 50 ? Math.min(1, (n - 50) / 50) : 0;
            integralSymbol.style.opacity = opacity;
            if (n >= 50) {
                const funcName = currentFuncExpr.includes('sin') ? 'sin(x)' :
                    currentFuncExpr.includes('x^2') ? 'x²' :
                    currentFuncExpr.includes('sqrt') ? '√x' :
                    currentFuncExpr.includes('exp') ? 'eˣ' : 'f(x)';
                integralSymbol.textContent = `∫_${a.toFixed(1)}^${b.toFixed(1)} ${funcName} dx = ${exactValue.toFixed(4)}`;
            }

            // 更新仪表盘
            updateDashboard(riemannData);
            // 更新误差指示器
            const absError = Math.abs(riemannData.sum - exactValue);
            errorIndicator.textContent = `|E| = ${absError.toFixed(6)}`;
            errorIndicator.style.color = absError < 0.01 ? 'var(--accent-green)' : 'var(--accent-red)';
            nOverlay.textContent = `n = ${n}`;

            // 绘制残差子图
            drawResidual(errors, dx);
        }

        function drawResidual(errors, dx) {
            const rect = residualCanvas.getBoundingClientRect();
            const W = rect.width;
            const H = rect.height;
            const dpr = window.devicePixelRatio || 1;

            if (residualCanvas.width !== W * dpr || residualCanvas.height !== H * dpr) {
                residualCanvas.width = W * dpr;
                residualCanvas.height = H * dpr;
                residualCanvas.style.width = W + 'px';
                residualCanvas.style.height = H + 'px';
                residualCtx.setTransform(1, 0, 0, 1, 0, 0);
                residualCtx.scale(dpr, dpr);

            ctx.clearRect(0, 0, W, H);
            var isLight = document.documentElement.getAttribute('data-theme') === 'light';
            ctx.fillStyle = isLight2 ? '#f0f0f0' : '#0a0a18';
            ctx.fillRect(0, 0, W, H);

            if (errors.length === 0) return;

            const maxErr = Math.max(...errors, 1e-10);
            const padding = { top: 15, bottom: 10, left: 10, right: 10 };
            const plotW = W - padding.left - padding.right;
            const plotH = H - padding.top - padding.bottom;

            // 绘制误差曲线
            ctx.beginPath();
            ctx.strokeStyle = '#ff8c42';
            ctx.lineWidth = 1.5;
            for (let i = 0; i < errors.length; i++) {
                const x = padding.left + (i / (errors.length - 1 || 1)) * plotW;
                const y = padding.top + plotH - (errors[i] / maxErr) * plotH;
                if (i === 0) ctx.moveTo(x, y);
                else ctx.lineTo(x, y);
            }
            ctx.stroke();

            // 填充下方
            ctx.lineTo(padding.left + plotW, padding.top + plotH);
            ctx.lineTo(padding.left, padding.top + plotH);
            ctx.closePath();
            ctx.fillStyle = 'rgba(255, 140, 66, 0.1)';
            ctx.fill();
        }

    // ==================== 编译函数 ====================
    function updateRFunc() {
        console.log('[polar] 编译表达式:', customExpr);
        rFunc = Common.compilePolarExpr(customExpr);
        if (rFunc) {
            try {
                const testVal = rFunc(0.5, paramK, paramA, paramPhi);
                console.log('[polar] 函数测试: r(0.5) =', testVal);
                if (isNaN(testVal) || !isFinite(testVal)) {
                    throw new Error('返回NaN或Infinity');
                }
                syntaxStatus.textContent = '✅ 有效方程';
                syntaxStatus.className = 'syntax-status syntax-ok';
            } catch (e) {
                console.error('[polar] 函数测试失败:', e);
                syntaxStatus.textContent = '❌ 运行时错误';
                syntaxStatus.className = 'syntax-status syntax-err';
                rFunc = null;
            }
        } else {
            syntaxStatus.textContent = '❌ 语法错误';
            syntaxStatus.className = 'syntax-status syntax-err';
            console.error('[polar] 编译失败:', customExpr);
        }
    }

    // ==================== 曲线生成 ====================
    function generatePolarCurve(steps) {
        steps = steps || 2000;
        const points = [];
        if (!rFunc) return points;
        const dTheta = (thetaEnd - thetaStart) / steps;
        for (let i = 0; i <= steps; i++) {
            const theta = thetaStart + i * dTheta;
            let r, valid = true;
            try {
                r = rFunc(theta, paramK, paramA, paramPhi);
                if (isNaN(r) || !isFinite(r) || Math.abs(r) > 1e6) valid = false;
            } catch (e) {
                r = 0;
                valid = false;
            }
            const x = valid ? r * Math.cos(theta) : 0;
            const y = valid ? r * Math.sin(theta) : 0;
            points.push({ theta, r, x, y, valid });
        }
        return points;
    }

    // ==================== 坐标转换 ====================
    function getPolarRect() { return polarCanvas.getBoundingClientRect(); }

    function polarToCanvas(mathX, mathY, W, H) {
        const scale = Math.min(W, H) / (2 * viewRange);
        const cx = W / 2 + viewCX * scale;
        const cy = H / 2 - viewCY * scale;
        return { x: cx + mathX * scale, y: cy - mathY * scale, scale: scale };
    }

    function canvasToPolar(canvasX, canvasY, W, H) {
        const scale = Math.min(W, H) / (2 * viewRange);
        const cx = W / 2 + viewCX * scale;
        const cy = H / 2 - viewCY * scale;
        return { x: (canvasX - cx) / scale, y: (cy - canvasY) / scale };
    }

    // ==================== 绘制极坐标视图 ====================
    function drawPolar() {
        const rect = getPolarRect();
        const W = rect.width, H = rect.height;
        const dpr = window.devicePixelRatio || 1;

        if (polarCanvas.width !== W * dpr || polarCanvas.height !== H * dpr) {
            polarCanvas.width = W * dpr;
            polarCanvas.height = H * dpr;
            polarCanvas.style.width = W + 'px';
            polarCanvas.style.height = H + 'px';
            polarCtx.setTransform(1, 0, 0, 1, 0, 0);
            polarCtx.scale(dpr, dpr);
        }

        const ctx = polarCtx;
        ctx.clearRect(0, 0, W, H);
        var isLight = document.documentElement.getAttribute('data-theme') === 'light';
        ctx.fillStyle = isLight ? '#ffffff' : '#0d1117';
        ctx.fillRect(0, 0, W, H);

        const curves = generatePolarCurve(isWeaving ? 500 : 2000);

        if (chkGrid.checked) drawPolarGrid(ctx, W, H);

        const rainbowMode = chkRainbow.checked;
        if (isWeaving) {
            drawPartialCurve(ctx, curves, weaveTheta, W, H, rainbowMode);
            const frontPt = getPointAtTheta(curves, weaveTheta);
            if (frontPt && frontPt.valid) {
                const pos = polarToCanvas(frontPt.x, frontPt.y, W, H);
                ctx.beginPath();
                ctx.fillStyle = '#ffd700';
                ctx.shadowColor = '#ffd700';
                ctx.shadowBlur = 16;
                ctx.arc(pos.x, pos.y, 5, 0, Math.PI * 2);
                ctx.fill();
                ctx.shadowBlur = 0;
            }
        } else {
            drawFullCurve(ctx, curves, W, H, rainbowMode);
        }

        if (chkProbe.checked && !isWeaving) drawProbe(ctx, curves, W, H);
        if (chkParticles.checked && !isWeaving) updateAnddrawParticles(ctx, curves, W, H);

        const origin = polarToCanvas(0, 0, W, H);
        ctx.beginPath();
        ctx.fillStyle = '#fff';
        ctx.arc(origin.x, origin.y, 3, 0, Math.PI * 2);
        ctx.fill();

        updateDashboard(curves);
    }

    function drawPolarGrid(ctx, W, H) {
        var isLight = document.documentElement.getAttribute('data-theme') === 'light';
        const origin = polarToCanvas(0, 0, W, H);
        const scale = Math.min(W, H) / (2 * viewRange);
        const maxR = viewRange * 1.5;

        for (let r = 0.5; r <= maxR; r += 0.5) {
            const canvasR = r * scale;
            ctx.beginPath();
            ctx.strokeStyle = isLight ? 'rgba(0,0,0,0.08)' : 'rgba(255,255,255,0.08)';
            ctx.lineWidth = 0.5;
            ctx.setLineDash([4, 6]);
            ctx.arc(origin.x, origin.y, canvasR, 0, Math.PI * 2);
            ctx.stroke();
            ctx.setLineDash([]);
            if (r <= viewRange + 0.1) {
                ctx.fillStyle = isLight ? 'rgba(0,0,0,0.5)' : 'rgba(255,255,255,0.3)';
                ctx.font = '8px "JetBrains Mono", monospace';
                ctx.fillText(r.toFixed(1), origin.x + canvasR + 3, origin.y - 3);
            }
        }

        const angles = [0, 30, 45, 60, 90, 120, 135, 150, 180, 210, 225, 240, 270, 300, 315, 330];
        const mainAngles = [0, 45, 90, 135, 180, 225, 270, 315];
        const rayLen = maxR * scale;

        angles.forEach(ang => {
            const rad = ang * Math.PI / 180;
            const x2 = origin.x + rayLen * Math.cos(rad);
            const y2 = origin.y - rayLen * Math.sin(rad);
            const isMain = mainAngles.includes(ang);
            const isAxis = (ang === 0 || ang === 180);

            ctx.beginPath();
            ctx.strokeStyle = isAxis ? (isLight ? 'rgba(0,0,0,0.3)' : 'rgba(255,255,255,0.3)') : isMain ? (isLight ? 'rgba(0,0,0,0.15)' : 'rgba(255,255,255,0.15)') : (isLight ? 'rgba(0,0,0,0.06)' : 'rgba(255,255,255,0.06)');
            ctx.lineWidth = isMain ? 0.8 : 0.4;
            ctx.setLineDash(isAxis ? [] : [3, 8]);
            ctx.moveTo(origin.x, origin.y);
            ctx.lineTo(x2, y2);
            ctx.stroke();
            ctx.setLineDash([]);

            const labelR = rayLen + 12;
            const lx = origin.x + labelR * Math.cos(rad);
            const ly = origin.y - labelR * Math.sin(rad);
            ctx.fillStyle = isLight ? 'rgba(0,0,0,0.5)' : 'rgba(255,255,255,0.35)';
            ctx.font = '8px "JetBrains Mono", monospace';
            ctx.fillText(ang + '°', lx - 10, ly + 3);
        });
    }

    function drawFullCurve(ctx, curves, W, H, rainbow) {
        if (!curves || curves.length < 2) return;
        const margin = 100;
        for (let i = 0; i < curves.length - 1; i++) {
            const p0 = curves[i];
            const p1 = curves[i + 1];
            if (!p0.valid || !p1.valid) continue;
            if (Math.abs(p0.r) > viewRange * 2.5 || Math.abs(p1.r) > viewRange * 2.5) continue;

            const pos0 = polarToCanvas(p0.x, p0.y, W, H);
            const pos1 = polarToCanvas(p1.x, p1.y, W, H);

            if ((pos0.x < -margin && pos1.x < -margin) || 
                (pos0.x > W + margin && pos1.x > W + margin) ||
                (pos0.y < -margin && pos1.y < -margin) || 
                (pos0.y > H + margin && pos1.y > H + margin)) continue;

            ctx.beginPath();
            ctx.lineWidth = rainbow ? 3 : 2.5;
            ctx.lineCap = 'round';

            if (rainbow) {
                const hue = ((p0.theta % (2 * Math.PI)) / (2 * Math.PI)) * 360;
                ctx.strokeStyle = 'hsl(' + hue + ', 90%, 60%)';
                ctx.shadowColor = 'hsl(' + hue + ', 90%, 60%)';
                ctx.shadowBlur = 4;
            } else {
                ctx.strokeStyle = '#ffd700';
                ctx.shadowColor = '#ffd700';
                ctx.shadowBlur = 3;
            }

            ctx.moveTo(pos0.x, pos0.y);
            ctx.lineTo(pos1.x, pos1.y);
            ctx.stroke();
            ctx.shadowBlur = 0;
        }
    }

    function drawPartialCurve(ctx, curves, maxTheta, W, H, rainbow) {
        if (curves.length < 2) return;
        for (let i = 0; i < curves.length - 1; i++) {
            const p0 = curves[i];
            const p1 = curves[i + 1];
            if (p0.theta > maxTheta) break;
            if (!p0.valid || !p1.valid) continue;
            if (Math.abs(p0.r) > viewRange * 3 || Math.abs(p1.r) > viewRange * 3) continue;

            const pos0 = polarToCanvas(p0.x, p0.y, W, H);
            const pos1 = polarToCanvas(p1.x, p1.y, W, H);

            ctx.beginPath();
            ctx.lineWidth = 3;
            ctx.lineCap = 'round';
            if (rainbow) {
                const hue = (p0.theta / (2 * Math.PI)) * 360;
                ctx.strokeStyle = 'hsl(' + hue + ', 90%, 60%)';
                ctx.shadowColor = 'hsl(' + hue + ', 90%, 60%)';
                ctx.shadowBlur = 4;
            } else {
                ctx.strokeStyle = '#ffd700';
                ctx.shadowColor = '#ffd700';
                ctx.shadowBlur = 3;
            }
            ctx.moveTo(pos0.x, pos0.y);
            ctx.lineTo(pos1.x, pos1.y);
            ctx.stroke();
            ctx.shadowBlur = 0;
        }
    }

    function getPointAtTheta(curves, targetTheta) {
        for (let i = curves.length - 1; i >= 0; i--) {
            if (curves[i].theta <= targetTheta && curves[i].valid) return curves[i];
        }
        return curves.length > 0 ? curves[0] : null;
    }

    function drawProbe(ctx, curves, W, H) {
        const origin = polarToCanvas(0, 0, W, H);
        const maxR = viewRange * 1.8;
        const scale = Math.min(W, H) / (2 * viewRange);

        const rad = probeTheta;
        const x2 = origin.x + maxR * scale * Math.cos(rad);
        const y2 = origin.y - maxR * scale * Math.sin(rad);
        ctx.beginPath();
        ctx.strokeStyle = 'rgba(72,201,176,0.6)';
        ctx.lineWidth = 1.5;
        ctx.setLineDash([6, 4]);
        ctx.moveTo(origin.x, origin.y);
        ctx.lineTo(x2, y2);
        ctx.stroke();
        ctx.setLineDash([]);

        const pt = getPointAtTheta(curves, probeTheta);
        if (pt && pt.valid) {
            const pos = polarToCanvas(pt.x, pt.y, W, H);
            ctx.beginPath();
            ctx.fillStyle = '#48c9b0';
            ctx.shadowColor = '#48c9b0';
            ctx.shadowBlur = 14;
            ctx.arc(pos.x, pos.y, 7, 0, Math.PI * 2);
            ctx.fill();
            ctx.shadowBlur = 0;
            ctx.strokeStyle = '#fff';
            ctx.lineWidth = 2;
            ctx.stroke();

            const handleDist = Math.min(pt.r * scale + 25, maxR * scale);
            const hx = origin.x + handleDist * Math.cos(rad);
            const hy = origin.y - handleDist * Math.sin(rad);
            ctx.beginPath();
            ctx.fillStyle = 'rgba(72,201,176,0.7)';
            ctx.arc(hx, hy, 10, 0, Math.PI * 2);
            ctx.fill();
            ctx.strokeStyle = '#fff';
            ctx.lineWidth = 2;
            ctx.stroke();

            angleOverlay.textContent = 'θ = ' + (probeTheta * 180 / Math.PI).toFixed(1) + '°';
            coordInfo.innerHTML = 'r = ' + pt.r.toFixed(3) + '<br>x = ' + pt.x.toFixed(3) + '<br>y = ' + pt.y.toFixed(3);
        }
    }

    // ==================== 粒子系统 ====================
    function updateAnddrawParticles(ctx, curves, W, H) {
        if (particles.length < MAX_PARTICLES && Math.random() < 0.3) {
            const theta = thetaStart + Math.random() * (thetaEnd - thetaStart);
            const r = rFunc ? rFunc(theta, paramK, paramA, paramPhi) : 0;
            if (isFinite(r) && Math.abs(r) < viewRange * 2) {
                particles.push({ theta, x: r * Math.cos(theta), y: r * Math.sin(theta), life: 1, speed: 0.002 + Math.random() * 0.008 });
            }
        }

        for (let i = particles.length - 1; i >= 0; i--) {
            const p = particles[i];
            p.theta += p.speed;
            p.life -= 0.003;
            if (p.theta > thetaEnd || p.life <= 0) { particles.splice(i, 1); continue; }
            const r = rFunc ? rFunc(p.theta, paramK, paramA, paramPhi) : 0;
            if (isFinite(r)) { p.x = r * Math.cos(p.theta); p.y = r * Math.sin(p.theta); }
            const pos = polarToCanvas(p.x, p.y, W, H);
            ctx.beginPath();
            ctx.fillStyle = 'rgba(255,215,0,' + p.life + ')';
            ctx.arc(pos.x, pos.y, 2, 0, Math.PI * 2);
            ctx.fill();
        }
    }

    function drawThetaR() {
        const rect = thetaRCanvas.getBoundingClientRect();

        const dpr = window.devicePixelRatio || 1;

        if (thetaRCanvas.width !== W * dpr || thetaRCanvas.height !== H * dpr) {

        const ctx = thetaRCtx;
        ctx.clearRect(0, 0, W, H);
        var isLight2 = document.documentElement.getAttribute('data-theme') === 'light';
        ctx.fillStyle = isLight2 ? '#f0f0f0' : '#0a0a18';
        ctx.fillRect(0, 0, W, H);

        const pad = { top: 25, bottom: 20, left: 40, right: 15 };
        const pw = W - pad.left - pad.right;
        const ph = H - pad.top - pad.bottom;
        const thetaMin = thetaStart, thetaMax = thetaEnd;
        const rMin = -viewRange * 1.5, rMax = viewRange * 1.5;

        ctx.strokeStyle = isLight2 ? '#ccc' : '#444';
        ctx.lineWidth = 1;
        const axisY = pad.top + ph / 2;
        ctx.beginPath();
        ctx.moveTo(pad.left, pad.top);
        ctx.lineTo(pad.left, pad.top + ph);
        ctx.moveTo(pad.left, axisY);
        ctx.lineTo(pad.left + pw, axisY);
        ctx.stroke();
        ctx.fillStyle = isLight2 ? '#666' : '#888';
        ctx.font = '9px "JetBrains Mono", monospace';
        ctx.fillText('θ', pad.left + pw - 12, axisY - 5);
        ctx.fillText('r', pad.left + 4, pad.top + 10);

        const steps = 500;
        ctx.beginPath();
        ctx.strokeStyle = '#ffd700';
        ctx.lineWidth = 1.5;
        let first = true;
        for (let i = 0; i <= steps; i++) {
            const theta = thetaMin + (thetaMax - thetaMin) * i / steps;
            const r = rFunc ? rFunc(theta, paramK, paramA, paramPhi) : 0;
            if (!isFinite(r)) { first = true; continue; }
            const px = pad.left + ((theta - thetaMin) / (thetaMax - thetaMin)) * pw;
            const py = pad.top + ph - ((r - rMin) / (rMax - rMin)) * ph;
            if (first) { ctx.moveTo(px, py); first = false; } else { ctx.lineTo(px, py); }
        }
        ctx.stroke();

        if (chkProbe.checked) {
            const pt = getPointAtTheta(generatePolarCurve(500), probeTheta);
            if (pt && pt.valid) {
                const px = pad.left + ((probeTheta - thetaMin) / (thetaMax - thetaMin)) * pw;
                const py = pad.top + ph - ((pt.r - rMin) / (rMax - rMin)) * ph;
                ctx.beginPath();
                ctx.fillStyle = '#48c9b0';
                ctx.shadowColor = '#48c9b0';
                ctx.shadowBlur = 10;
                ctx.arc(px, py, 5, 0, Math.PI * 2);
                ctx.fill();
                ctx.shadowBlur = 0;
            }
        }
    }

    function updateDashboard(curves) {
        const pt = getPointAtTheta(curves, probeTheta);
        if (pt && pt.valid) {
            document.getElementById('dashTheta').textContent = (probeTheta * 180 / Math.PI).toFixed(1) + '°';
            document.getElementById('dashR').textContent = pt.r.toFixed(4);

            } else {
                document.getElementById('dashPetals').textContent = '—';
                document.getElementById('dashType').textContent = '变形玫瑰线';
            }

            document.getElementById('dashPetals').textContent = '∞';
        }

        let length = 0, area = 0;
        for (let i = 0; i < curves.length - 1; i++) {
            const p0 = curves[i], p1 = curves[i + 1];
            if (!p0.valid || !p1.valid) continue;
            const dx = p1.x - p0.x, dy = p1.y - p0.y;
            length += Math.sqrt(dx * dx + dy * dy);
            area += 0.5 * Math.abs(p0.r * p1.r) * (p1.theta - p0.theta);
        }
        document.getElementById('dashLength').textContent = length.toFixed(2);
        document.getElementById('dashArea').textContent = area.toFixed(4);
    }

    function animateWeave() {
        if (!isWeaving) return;
        const speeds = [0, 0.03, 0.06, 0.12, 0.25];
        weaveTheta += speeds[weaveSpeed];
        if (weaveTheta >= thetaEnd) { weaveTheta = thetaEnd; stopWeave(); draw(); return; }
        draw();
        weaveId = requestAnimationFrame(animateWeave);

        if (weaveId) cancelAnimationFrame(weaveId);

        draw();

        const factorialCache = [1];
        function factorial(n) {
            if (n < factorialCache.length) return factorialCache[n];
            for (let i = factorialCache.length; i <= n; i++) {
                factorialCache.push(factorialCache[i - 1] * i);

        // ==================== 符号导数（针对预设函数） ====================
        function getDerivativeValue(symbolic, n, x) {
            switch (symbolic) {
                case 'sin':
                    const remSin = n % 4;
                    if (remSin === 0) return Math.sin(x);
                    if (remSin === 1) return Math.cos(x);
                    if (remSin === 2) return -Math.sin(x);
                    return -Math.cos(x);
                case 'cos':
                    const remCos = n % 4;
                    if (remCos === 0) return Math.cos(x);
                    if (remCos === 1) return -Math.sin(x);
                    if (remCos === 2) return -Math.cos(x);
                    return Math.sin(x);
                case 'exp':
                    return Math.exp(x);
                case 'ln1p':
                    // f(x)=ln(1+x), f^(n)(x) = (-1)^(n-1) * (n-1)! / (1+x)^n  (n>=1)
                    if (n === 0) return Math.log(1 + x);
                    return (Math.pow(-1, n - 1) * factorial(n - 1)) / Math.pow(1 + x, n);
                case 'geo':
                    // f(x)=1/(1-x), f^(n)(x) = n! / (1-x)^(n+1)
                    return factorial(n) / Math.pow(1 - x, n + 1);
                case 'atan':
                    // arctan的导数有规律，使用数值微分
                    const h = 1e-7;
                    if (n === 0) return Math.atan(x);
                    // 递归数值微分不稳定，使用解析公式近似
                    // f^(n)(arctan) at 0: 0 如果 n 偶数，(-1)^((n-1)/2)*(n-1)! 如果 n 奇数
                    if (Math.abs(x) < 1e-9) {

                        const k = (n - 1) / 2;
                        return Math.pow(-1, k) * factorial(n - 1);
                    }
                    // 一般情况用数值微分
                    return numericalDerivativeN(symbolic, n, x);
                default:
                    return numericalDerivativeN(symbolic, n, x);
            }
        }

        function numericalDerivativeN(symbolic, order, x) {
            const h = 1e-6;
            if (order === 0) return currentFunc(x);
            // 中心差分
            return (getDerivativeValue(symbolic, order - 1, x + h) -
                getDerivativeValue(symbolic, order - 1, x - h)) / (2 * h);
        }

        // ==================== 泰勒系数计算 ====================
        function computeTaylorCoefficients(symbolic, a, maxOrder) {
            const coeffs = [];
            for (let n = 0; n <= maxOrder; n++) {
                const derivAtA = getDerivativeValue(symbolic, n, a);
                coeffs.push(derivAtA / factorial(n));
            }
            return coeffs;
        }

        function hornerEvaluate(coeffs, x, a) {
            let result = 0;
            for (let i = coeffs.length - 1; i >= 0; i--) {
                result = result * (x - a) + coeffs[i];
            }
            return result;
        }

        // ==================== 生成泰勒表达式字符串 ====================
        function generateTaylorExpression(coeffs, a, n) {
            if (n < 0) return 'T₀(x) = 0';
            let expr = `T${n}(x) = `;
            let hasTerm = false;
            for (let k = 0; k <= n; k++) {
                const c = coeffs[k];
                // 根据stepping过滤
                if (stepping === 'odd' && k % 2 === 0) continue;
                if (stepping === 'even' && k % 2 === 1) continue;
                if (Math.abs(c) < 1e-15) continue;

                // 格式化系数
                let term = '';
                const absC = Math.abs(c);
                const isFrac = Math.abs(absC - Math.round(absC)) > 1e-10 && absC < 1;

                if (k === 0) {
                    term = c.toFixed(4);
                } else {
                    // 系数部分
                    let coeffStr = '';
                    if (Math.abs(absC - 1) < 1e-10) {
                        coeffStr = '';
                    } else if (isFrac) {
                        // 尝试显示为分数
                        const frac = toFraction(c);
                        coeffStr = frac;
                    } else {
                        coeffStr = parseFloat(c.toFixed(6)).toString();
                    }
                    // 变量部分
                    const aStr = Math.abs(a) < 1e-10 ? '' : `(x${a >= 0 ? '-' : '+'}${Math.abs(a).toFixed(2)})`;
                    const varPart = aStr || 'x';
                    const powPart = k === 1 ? varPart : `${varPart}^${k}`;
                    term = coeffStr === '' ? powPart : `${coeffStr}·${powPart}`;
                }

                if (!hasTerm) {
                    expr += (c < 0 ? '−' : '') + term;
                } else {
                    expr += (c < 0 ? ' − ' : ' + ') + term;
                }
                hasTerm = true;
            }
            if (!hasTerm) expr += '0';
            return expr;
        }

        function toFraction(c) {
            const absC = Math.abs(c);
            for (let denom = 2; denom <= 1000; denom++) {
                const num = Math.round(absC * denom);
                if (Math.abs(num / denom - absC) < 1e-10) {
                    return (c < 0 ? '−' : '') + `${num}/${denom}`;
                }
            }
            return parseFloat(c.toFixed(6)).toString();
        }

            const neededW = W * dpr;
            const neededH = graphH * dpr;
            if (graphCanvas.width !== neededW || graphCanvas.height !== neededH) {
                graphCanvas.width = neededW;
                graphCanvas.height = neededH;
                graphCanvas.style.width = W + 'px';
                graphCanvas.style.height = graphH + 'px';
                graphCtx.setTransform(1, 0, 0, 1, 0, 0);
                graphCtx.scale(dpr, dpr);
            }

            const ctx = graphCtx;
            ctx.clearRect(0, 0, W, graphH);
            var isLight = document.documentElement.getAttribute('data-theme') === 'light';
            ctx.fillStyle = isLight ? '#ffffff' : '#0d1117';
            ctx.fillRect(0, 0, W, graphH);

            // 网格
            ctx.strokeStyle = isLight ? '#e0e0e0' : '#1a1a30';
            ctx.lineWidth = 0.5;
            const gridStepX = Math.pow(10, Math.floor(Math.log10(xMax - xMin))) / 5;
            const gridStepY = Math.pow(10, Math.floor(Math.log10(yMax - yMin))) / 5;
            for (let x = Math.floor(xMin / gridStepX) * gridStepX; x <= xMax; x += gridStepX) {
                ctx.beginPath();
                ctx.moveTo(toCanvasX(x, W), 0);
                ctx.lineTo(toCanvasX(x, W), graphH);
                ctx.stroke();
            }
            for (let y = Math.floor(yMin / gridStepY) * gridStepY; y <= yMax; y += gridStepY) {
                ctx.beginPath();
                ctx.moveTo(0, toCanvasY(y, graphH));
                ctx.lineTo(W, toCanvasY(y, graphH));
                ctx.stroke();
            }

            // 坐标轴
            const axisY = toCanvasY(0, graphH);
            const axisX = toCanvasX(0, W);
            ctx.strokeStyle = isLight ? '#bbb' : '#888';
            ctx.lineWidth = 1.5;
            ctx.beginPath();
            ctx.moveTo(0, axisY);
            ctx.lineTo(W, axisY);
            ctx.moveTo(axisX, 0);
            ctx.lineTo(axisX, graphH);
            ctx.stroke();
            ctx.fillStyle = isLight ? '#666' : '#aaa';
            ctx.font = '11px "Segoe UI", sans-serif';
            ctx.fillText('x', W - 16, axisY - 8);
            ctx.fillText('y', axisX + 10, 14);

            // 收敛半径
            const alphaRadius = chkRadius.checked;
            if (alphaRadius && (symbolicType === 'ln1p' || symbolicType === 'geo')) {
                const R = symbolicType === 'ln1p' ? 1 : 1;
                const cxR = toCanvasX(expansionPoint + R, W);
                const cxL = toCanvasX(expansionPoint - R, W);
                ctx.strokeStyle = 'rgba(255,80,80,0.5)';
                ctx.lineWidth = 2;
                ctx.setLineDash([10, 5]);
                ctx.beginPath();
                ctx.moveTo(cxR, 0);
                ctx.lineTo(cxR, graphH);
                ctx.moveTo(cxL, 0);
                ctx.lineTo(cxL, graphH);
                ctx.stroke();
                ctx.setLineDash([]);
                ctx.fillStyle = 'rgba(255,80,80,0.7)';
                ctx.font = '10px "Segoe UI", sans-serif';
                ctx.fillText(`R=1`, cxR + 3, graphH - 8);
            }

            // 系数
            const maxOrder = Math.min(maxN, 15);
            const coeffs = computeTaylorCoefficients(symbolicType, expansionPoint, maxOrder);

            // 历史快照
            const showHistory = chkHistory.checked;
            const historyColors = [
                'rgba(100,150,255,0.4)', // n=1
                'rgba(120,200,180,0.45)', // n=3
                'rgba(200,160,80,0.5)', // n=5
                'rgba(255,140,60,0.55)', // n=7
            ];
            if (showHistory) {
                const historyNs = [1, 3, 5, 7].filter(n => n <= maxOrder);
                historyNs.forEach((hN, idx) => {
                    const histCoeffs = computeTaylorCoefficients(symbolicType, expansionPoint, hN);
                    drawPolynomialCurve(ctx, histCoeffs, expansionPoint, W, graphH,
                        historyColors[idx], 1.5, [2, 4]);
                    // 标注
                    const labelX = toCanvasX(expansionPoint + 0.8, W);
                    const labelY = toCanvasY(hornerEvaluate(histCoeffs, expansionPoint + 0.8,
                        expansionPoint), graphH);
                    ctx.fillStyle = historyColors[idx].replace('0.4', '0.9');
                    ctx.font = 'bold 10px var(--mono)';
                    ctx.fillText(`n=${hN}`, labelX, labelY - 6);
                });
            }

            // 泰勒多项式（金色）
            const effectiveN = getEffectiveN(maxOrder);
            const goldAlpha = 0.55 + (effectiveN / 15) * 0.45;
            const goldR = Math.round(255 * (0.6 + effectiveN / 15 * 0.4));
            const goldG = Math.round(215 * (0.5 + effectiveN / 15 * 0.5));
            const goldB = Math.round(0);
            const goldColor = `rgba(${goldR},${goldG},${goldB},${goldAlpha})`;
            const goldWidth = 1.5 + (effectiveN / 15) * 2.5;
            const goldDash = effectiveN >= 10 ? [] : [6, 3];
            drawPolynomialCurve(ctx, coeffs, expansionPoint, W, graphH, goldColor, goldWidth, goldDash);

            // 原函数（白色发光）
            ctx.save();
            ctx.shadowColor = 'rgba(255,255,255,0.4)';
            ctx.shadowBlur = 6;
            ctx.beginPath();
            ctx.strokeStyle = '#ffffff';
            ctx.lineWidth = 3.5;
            ctx.lineJoin = 'round';
            let first = true;
            for (let px = 0; px <= W; px += 1.5) {
                const mathX = toMathX(px, W);
                const mathY = currentFunc(mathX);
                if (isNaN(mathY) || !isFinite(mathY) || Math.abs(mathY) > 1e6) { first = true; continue; }
                const cy = toCanvasY(mathY, graphH);
                if (cy < -500 || cy > graphH + 500) { first = true; continue; }
                if (first) { ctx.moveTo(px, cy);
                    first = false; } else { ctx.lineTo(px, cy); }
            }
            ctx.stroke();
            ctx.restore();

            // 展开点标记
            const epCanvasX = toCanvasX(expansionPoint, W);
            const epCanvasY = toCanvasY(currentFunc(expansionPoint), graphH);
            ctx.beginPath();
            ctx.fillStyle = '#ff3333';
            ctx.shadowColor = '#ff3333';
            ctx.shadowBlur = 12;
            ctx.arc(epCanvasX, epCanvasY, 7, 0, Math.PI * 2);
            ctx.fill();
            ctx.shadowBlur = 0;
            ctx.strokeStyle = '#fff';
            ctx.lineWidth = 2;
            ctx.stroke();
            // 脉冲光晕
            ctx.beginPath();
            ctx.fillStyle = 'rgba(255,50,50,0.25)';
            ctx.arc(epCanvasX, epCanvasY, 14, 0, Math.PI * 2);
            ctx.fill();

            // 鼠标十字线与误差棒
            if (mouseCanvasX !== null && mouseY !== null && graphH > 0) {
                const mx = mouseCanvasX;
                const mathX = toMathX(mx, W);
                const fVal = currentFunc(mathX);
                const tVal = hornerEvaluate(coeffs, mathX, expansionPoint);
                if (isFinite(fVal) && isFinite(tVal)) {
                    const fCanvasY = toCanvasY(fVal, graphH);
                    const tCanvasY = toCanvasY(tVal, graphH);
                    // 十字线
                    ctx.strokeStyle = isLight ? 'rgba(0,0,0,0.15)' : 'rgba(255,255,255,0.15)';
                    ctx.lineWidth = 1;
                    ctx.setLineDash([4, 6]);
                    ctx.beginPath();
                    ctx.moveTo(mx, 0);
                    ctx.lineTo(mx, graphH);
                    ctx.stroke();
                    ctx.setLineDash([]);
                    // 误差棒
                    if (chkErrorBar.checked) {
                        ctx.beginPath();
                        ctx.strokeStyle = '#e05161';
                        ctx.lineWidth = 2;
                        ctx.moveTo(mx + 8, Math.min(fCanvasY, tCanvasY));
                        ctx.lineTo(mx + 8, Math.max(fCanvasY, tCanvasY));
                        ctx.stroke();
                        ctx.fillStyle = '#fff';
                        ctx.beginPath();
                        ctx.arc(mx + 8, Math.min(fCanvasY, tCanvasY), 3, 0, Math.PI * 2);
                        ctx.fill();
                        ctx.beginPath();
                        ctx.arc(mx + 8, Math.max(fCanvasY, tCanvasY), 3, 0, Math.PI * 2);
                        ctx.fill();
                        // 工具提示
                        const error = Math.abs(fVal - tVal);
                        errorTooltip.style.display = 'block';
                        errorTooltip.style.left = (mx + 16) + 'px';
                        errorTooltip.style.top = (Math.min(fCanvasY, tCanvasY) - 22) + 'px';
                        errorTooltip.textContent = `|E|= ${error.toExponential(3)}`;
                    }
                    // 光标信息
                    cursorInfo.innerHTML =
                        `f(x) = ${fVal.toFixed(6)}<br>Tₙ(x) = ${tVal.toFixed(6)}<br>|E| = ${Math.abs(fVal-tVal).toExponential(3)}`;
                }
            }

            // 更新仪表盘
            updateDashboard(coeffs);
            updateExpressionDisplay(coeffs);
            nOverlay.textContent = `n = ${getEffectiveN(maxOrder)}`;
            dashN.textContent = getEffectiveN(maxOrder);

            // 绘制热力图
            if (chkHeatmap.checked) {
                drawHeatmap(coeffs, W);
                heatmapPanel.style.display = 'block';
            } else {
                heatmapPanel.style.display = 'none';
                heatmapCanvas.style.display = 'none';
            }
            if (chkHeatmap.checked) {
                heatmapCanvas.style.display = 'block';
            }
        }

        function drawPolynomialCurve(ctx, coeffs, a, W, graphH, color, width, dash) {
            ctx.beginPath();
            ctx.strokeStyle = color;
            ctx.lineWidth = width;
            ctx.lineJoin = 'round';
            if (dash && dash.length > 0) ctx.setLineDash(dash);
            let first = true;
            for (let px = 0; px <= W; px += 1.5) {
                const mathX = toMathX(px, W);
                const mathY = hornerEvaluate(coeffs, mathX, a);
                if (isNaN(mathY) || !isFinite(mathY) || Math.abs(mathY) > 1e6) { first = true; continue; }
                const cy = toCanvasY(mathY, graphH);
                if (cy < -500 || cy > graphH + 500) { first = true; continue; }
                if (first) { ctx.moveTo(px, cy);
                    first = false; } else { ctx.lineTo(px, cy); }
            }
            ctx.stroke();
            ctx.setLineDash([]);
        }

        function getEffectiveN(maxOrder) {
            if (stepping === 'all') return maxOrder;
            if (stepping === 'odd') return maxOrder % 2 === 0 ? Math.max(0, maxOrder - 1) : maxOrder;
            if (stepping === 'even') return maxOrder % 2 === 1 ? Math.max(0, maxOrder - 1) : maxOrder;
            return maxOrder;
        }

        // ==================== 热力图 ====================
        function drawHeatmap(coeffs, graphW) {
            const rect = heatmapCanvas.getBoundingClientRect();
            const W = rect.width;
            const H = rect.height;
            const dpr = window.devicePixelRatio || 1;

            if (heatmapCanvas.width !== W * dpr || heatmapCanvas.height !== H * dpr) {
                heatmapCanvas.width = W * dpr;
                heatmapCanvas.height = H * dpr;
                heatmapCanvas.style.width = W + 'px';
                heatmapCanvas.style.height = H + 'px';
                heatmapCtx.setTransform(1, 0, 0, 1, 0, 0);
                heatmapCtx.scale(dpr, dpr);
            }

            const ctx = heatmapCtx;
            const imgData = ctx.createImageData(W, H);
            for (let px = 0; px < W; px++) {
                const mathX = toMathX(px * (graphW / W), graphW);
                const fVal = currentFunc(mathX);
                const tVal = hornerEvaluate(coeffs, mathX, expansionPoint);
                let error = Math.abs(fVal - tVal);
                if (!isFinite(error)) error = 1e6;
                // 对数尺度映射颜色
                const logError = Math.log10(Math.max(error, 1e-15));
                // 映射: -15→深蓝, -5→浅蓝, 0→白, 3→红, 6→深红
                const t = (logError + 15) / 21; // 归一化到 0~1
                const clampedT = Math.max(0, Math.min(1, t));
                const [r, g, b] = 	Common.heatmapColor(clampedT);
                for (let py = 0; py < H; py++) {
                    const idx = (py * W + px) * 4;
                    imgData.data[idx] = r;
                    imgData.data[idx + 1] = g;
                    imgData.data[idx + 2] = b;
                    imgData.data[idx + 3] = 255;
                }
            }
            ctx.putImageData(imgData, 0, 0);
        }

        

        // ==================== 仪表盘更新 ====================
        function updateDashboard(coeffs) {
            let maxError = 0;
            const steps = 500;
            for (let i = 0; i <= steps; i++) {
                const x = xMin + (xMax - xMin) * i / steps;
                const fVal = currentFunc(x);
                const tVal = hornerEvaluate(coeffs, x, expansionPoint);
                if (isFinite(fVal) && isFinite(tVal)) {
                    maxError = Math.max(maxError, Math.abs(fVal - tVal));
                }
            }
            dashMaxError.textContent = maxError.toExponential(3);
            if (maxError > 0.1) dashMaxError.className = 'dash-value dash-red';
            else if (maxError < 0.001) dashMaxError.className = 'dash-value dash-green';
            else dashMaxError.className = 'dash-value';

            // 鼠标处误差
            if (mouseCanvasX !== null) {
                const { W } = getGraphRect();
                const mathX = toMathX(mouseCanvasX, W);
                const fVal = currentFunc(mathX);
                const tVal = hornerEvaluate(coeffs, mathX, expansionPoint);
                if (isFinite(fVal) && isFinite(tVal)) {
                    const err = Math.abs(fVal - tVal);
                    dashError.textContent = err.toExponential(3);
                    dashError.className = err < 0.001 ? 'dash-value dash-green' : 'dash-value dash-red';
                }
            }
        }

        function updateExpressionDisplay(coeffs) {
            const effectiveN = getEffectiveN(maxN);
            const expr = generateTaylorExpression(coeffs, expansionPoint, effectiveN);
            exprOverlay.textContent = expr.replace(/T\d+\(x\) = /, '');
            taylorExprDisplay.textContent = expr;
            exprOverlay.textContent = expr.length > 50 ? expr.substring(0, 47) + '...' : expr;
        }

        // ==================== 交互事件 ====================
        graphCanvas.addEventListener('pointerdown', function(e) {
            const rect = getGraphRect();
            const W = rect.W;
            const graphH = rect.H;
            const mx = e.clientX - graphCanvas.getBoundingClientRect().left;
            const my = e.clientY - graphCanvas.getBoundingClientRect().top;

            const epCanvasX = toCanvasX(expansionPoint, W);
            const epCanvasY = toCanvasY(currentFunc(expansionPoint), graphH);
            const dist = Math.hypot(mx - epCanvasX, my - epCanvasY);

            if (dist < 16) {
                draggingExpansionPoint = true;
                graphCanvas.style.cursor = 'grabbing';
                e.preventDefault();
                return;
            }

            if (e.button === 2 || (e.button === 0 && e.ctrlKey)) {
                panning = true;
                panStart = { x: e.clientX, y: e.clientY, xMin, xMax, yMin, yMax };
                graphCanvas.style.cursor = 'grabbing';
                e.preventDefault();
                return;
            }
            mouseCanvasX = mx;
            mouseY = my;
            draw();
        });

        window.addEventListener('pointermove', function(e) {
            const rect = graphCanvas.getBoundingClientRect();
            const mx = e.clientX - rect.left;
            const my = e.clientY - rect.top;
            const { W, H } = getGraphRect();
            const graphH = H;

            if (draggingExpansionPoint) {
                expansionPoint = Math.max(xMin, Math.min(xMax, toMathX(mx, W)));
                inputA.value = parseFloat(expansionPoint.toFixed(2));
                draw();
                return;
            }
            if (panning && panStart) {
                const dx = (e.clientX - panStart.x) / W * (panStart.xMax - panStart.xMin);
                const dy = (e.clientY - panStart.y) / graphH * (panStart.yMax - panStart.yMin);
                xMin = panStart.xMin - dx;
                xMax = panStart.xMax - dx;
                yMin = panStart.yMin + dy;
                yMax = panStart.yMax + dy;
                draw();
                return;
            }
            mouseCanvasX = mx;
            mouseY = my;
            draw();
        });

        window.addEventListener('pointerup', function() {
            draggingExpansionPoint = false;
            panning = false;
            panStart = null;
            graphCanvas.style.cursor = 'crosshair';
        });

        function copyLatex() {
            const coeffs = computeTaylorCoefficients(symbolicType, expansionPoint,
                Math.min(maxN, 15));
            const effN = getEffectiveN(maxN);
            let latex = `T_{${effN}}(x) = `;
            let hasTerm = false;
            for (let k = 0; k <= effN; k++) {
                if (stepping === 'odd' && k % 2 === 0) continue;
                if (stepping === 'even' && k % 2 === 1) continue;
                const c = coeffs[k];
                if (Math.abs(c) < 1e-15) continue;
                const sign = c < 0 ? ' - ' : (hasTerm ? ' + ' : '');
                const absC = Math.abs(c);
                let term = '';
                if (k === 0) {
                    term = parseFloat(c.toFixed(6)).toString();
                } else {
                    const coeffFrac = toLatexFraction(c);
                    const aStr = Math.abs(expansionPoint) < 1e-10 ? 'x' :
                        `(x ${expansionPoint >= 0 ? '-' : '+'} ${Math.abs(expansionPoint).toFixed(2)})`;
                    const powPart = k === 1 ? aStr : `${aStr}^{${k}}`;
                    term = coeffFrac === '1' ? powPart :
                        coeffFrac === '-1' ? `-${powPart}` : `${coeffFrac}\\,${powPart}`;
                }
                latex += sign + term;
                hasTerm = true;
            }
            if (!hasTerm) latex += '0';
            navigator.clipboard.writeText(latex).then(() => {
                const btn = document.querySelector('.btn-latex');
                const orig = btn.textContent;
                btn.textContent = '✅ 已复制!';
                setTimeout(() => { btn.textContent = orig; }, 1500);
            });
        }

        function toLatexFraction(c) {
            const absC = Math.abs(c);
            if (Math.abs(absC - 1) < 1e-10) return c < 0 ? '-1' : '1';
            for (let denom = 2; denom <= 1000; denom++) {
                const num = Math.round(absC * denom);
                if (Math.abs(num / denom - absC) < 1e-10) {
                    return (c < 0 ? '-' : '') + `\\frac{${num}}{${denom}}`;
                }
            }
            return parseFloat(c.toFixed(6)).toString();
        }

        ctx.clearRect(0,0,W,H);
        ctx.fillStyle = isLight ? '#f0f2f5' : '#0d1117';
        ctx.fillRect(0,0,W,H);

        if (!binaryFunc) return;
        if (projectedGrid.length === 0) reprojectAll();

        var absMax = Math.max(Math.abs(globalMaxZ), Math.abs(globalMinZ), 0.1);

        // 参考平面
        if (chkGround.checked) drawGroundPlane(W, H, isLight);

        // 曲面
        if (chkFill.checked) drawFilledSurface(absMax);
        if (chkWire.checked) drawWireframe(absMax);

        // 坐标轴
        if (chkAxes.checked) drawAxes(W, H, isLight);

        updateAngleDisplay();
    }

    function drawGroundPlane(W, H, isLight) {
        var cc = getCanvasCenter();
        var corners = [
            project(xMin, yMin, 0), project(xMax, yMin, 0),
            project(xMax, yMax, 0), project(xMin, yMax, 0)
        ];
        ctx.beginPath();
        ctx.moveTo(corners[0].sx, corners[0].sy);
        for (var i = 1; i < 4; i++) ctx.lineTo(corners[i].sx, corners[i].sy);
        ctx.closePath();
        ctx.fillStyle = isLight ? 'rgba(200,210,220,0.15)' : 'rgba(255,255,255,0.03)';
        ctx.fill();
        ctx.strokeStyle = isLight ? 'rgba(0,0,0,0.2)' : 'rgba(255,255,255,0.1)';
        ctx.lineWidth = 1;
        ctx.setLineDash([4, 6]);
        ctx.stroke();
        ctx.setLineDash([]);
    }

    function drawFilledSurface(absMax) {
        var rows = projectedGrid.length;
        if (rows < 2) return;
        var cols = projectedGrid[0].length;
        var sortedQuads = [];
        for (var i = 0; i < rows - 1; i++) {
            for (var j = 0; j < cols - 1; j++) {
                var p00 = projectedGrid[i][j], p10 = projectedGrid[i][j+1];
                var p01 = projectedGrid[i+1][j], p11 = projectedGrid[i+1][j+1];
                if (!isFinite(p00.z3) || !isFinite(p10.z3) || !isFinite(p01.z3) || !isFinite(p11.z3)) continue;
                var avgDepth = (p00.depth + p10.depth + p01.depth + p11.depth) / 4;
                var avgZ = (p00.z3 + p10.z3 + p01.z3 + p11.z3) / 4;
                sortedQuads.push({
                    i: i, j: j, depth: avgDepth, avgZ: avgZ,
                    p00: p00, p10: p10, p01: p01, p11: p11
                });
            }
        }
        sortedQuads.sort(function(a, b) { return b.depth - a.depth; });

        for (var k = 0; k < sortedQuads.length; k++) {
            var q = sortedQuads[k];
            var nz = q.avgZ / absMax;
            var color = Common.bidirectionalColor(nz);
            ctx.beginPath();
            ctx.moveTo(q.p00.sx, q.p00.sy);
            ctx.lineTo(q.p10.sx, q.p10.sy);
            ctx.lineTo(q.p11.sx, q.p11.sy);
            ctx.lineTo(q.p01.sx, q.p01.sy);
            ctx.closePath();
            ctx.fillStyle = 'rgba(' + color[0] + ',' + color[1] + ',' + color[2] + ',0.25)';
            ctx.fill();
        }
    }

    function drawWireframe(absMax) {
        var rows = projectedGrid.length;
        if (rows < 2) return;
        var cols = projectedGrid[0].length;
        ctx.lineWidth = 0.7;

        // 先画纵向线(从远到近)
        for (var i = 0; i < rows; i++) {
            for (var j = 0; j < cols - 1; j++) {
                var p0 = projectedGrid[i][j], p1 = projectedGrid[i][j+1];
                if (!isFinite(p0.z3) || !isFinite(p1.z3)) continue;
                var nz = (p0.z3 + p1.z3) / 2 / absMax;
                var color = Common.bidirectionalColor(nz);
                ctx.beginPath();
                ctx.strokeStyle = 'rgba(' + color[0] + ',' + color[1] + ',' + color[2] + ',0.6)';
                ctx.moveTo(p0.sx, p0.sy); ctx.lineTo(p1.sx, p1.sy);
                ctx.stroke();
            }
        }
        // 再画横向线
        for (var j = 0; j < cols; j++) {
            for (var i = 0; i < rows - 1; i++) {
                var p0 = projectedGrid[i][j], p1 = projectedGrid[i+1][j];
                if (!isFinite(p0.z3) || !isFinite(p1.z3)) continue;
                var nz = (p0.z3 + p1.z3) / 2 / absMax;
                var color = Common.bidirectionalColor(nz);
                ctx.beginPath();
                ctx.strokeStyle = 'rgba(' + color[0] + ',' + color[1] + ',' + color[2] + ',0.5)';
                ctx.moveTo(p0.sx, p0.sy); ctx.lineTo(p1.sx, p1.sy);
                ctx.stroke();
            }
        }
    }

    function drawAxes(W, H, isLight) {
        var o = project(0, 0, 0);
        var axisLen = Math.max(xMax-xMin, yMax-yMin) * 0.6;
        var xEnd = project(axisLen, 0, 0);
        var yEnd = project(0, axisLen, 0);
        var zEnd = project(0, 0, axisLen);

        var colors = [
            { end: xEnd, color: '#e05161', label: 'X' },
            { end: yEnd, color: '#4caf84', label: 'Y' },
            { end: zEnd, color: '#4f9fcf', label: 'Z' }
        ];

        colors.forEach(function(a) {
            ctx.beginPath();
            ctx.strokeStyle = a.color;
            ctx.lineWidth = 2;
            ctx.moveTo(o.sx, o.sy);
            ctx.lineTo(a.end.sx, a.end.sy);
            ctx.stroke();
            ctx.fillStyle = a.color;
            ctx.font = 'bold 11px sans-serif';
            ctx.fillText(a.label, a.end.sx + 5, a.end.sy - 3);
        });

        // 原点球
        ctx.beginPath();
        ctx.fillStyle = '#fff';
        ctx.arc(o.sx, o.sy, 4, 0, Math.PI*2);
        ctx.fill();
    }

    function updateAngleDisplay() {
        var pitchDeg = Math.round(angleX * 180 / Math.PI);
        var yawDeg = Math.round(angleZ * 180 / Math.PI);
        angleOverlay.textContent = '俯仰' + pitchDeg + '° 旋转' + yawDeg + '°';
        document.getElementById('pitchValue').textContent = pitchDeg + '°';
        document.getElementById('yawValue').textContent = yawDeg + '°';
    }

    // ==================== 交互 ====================
    canvas.addEventListener('pointerdown', function(e) {
        if (autoRotating) stopAutoRotate();
        if (e.button === 2 || (e.button === 0 && e.ctrlKey)) {
            isPanning = true;
            panStartCX = centerX - e.clientX;
            panStartCY = centerY - e.clientY;
            container.style.cursor = 'grabbing';
            e.preventDefault(); return;
        }
        if (e.button === 0) {
            isDragging = true;
            dragStartX = e.clientX; dragStartY = e.clientY;
            dragStartAX = angleX; dragStartAZ = angleZ;
            velocityX = 0; velocityZ = 0;
            container.style.cursor = 'grabbing';
            e.preventDefault();
        }
    });

    window.addEventListener('pointermove', function(e) {
        if (isDragging) {
            var dx = e.clientX - dragStartX;
            var dy = e.clientY - dragStartY;
            angleZ = dragStartAZ - dx * 0.0012;
            angleX = dragStartAX + dy * 0.0012;
            angleX = Math.max(-Math.PI/2.2, Math.min(Math.PI/2.2, angleX));
            updateAngleSliders();
            reprojectAll(); draw();
            return;
        }
        if (isPanning) {
            centerX = panStartCX + e.clientX;
            centerY = panStartCY + e.clientY;
            reprojectAll(); draw();
            return;
        }
    });

    window.addEventListener('pointerup', function(e) {
        if (isDragging) {
            velocityX = (dragStartAX - angleX) * 1.5;
            velocityZ = (dragStartAZ - angleZ) * 1.5;
            isDragging = false;
            container.style.cursor = 'grab';
            if (Math.abs(velocityX) > 0.001 || Math.abs(velocityZ) > 0.001) {
                animateInertia();
            }
        }
        if (isPanning) { isPanning = false; container.style.cursor = 'grab'; }
    });

    function animateInertia() {
        if (Math.abs(velocityX) < 0.0005 && Math.abs(velocityZ) < 0.0005) return;
        angleX -= velocityX;
        angleZ -= velocityZ;
        angleX = Math.max(-Math.PI/2.2, Math.min(Math.PI/2.2, angleX));
        velocityX *= 0.90;
        velocityZ *= 0.90;
        updateAngleSliders();
        reprojectAll(); draw();
        requestAnimationFrame(animateInertia);
    }

    canvas.addEventListener('wheel', function(e) {
        e.preventDefault();
        var factor = e.deltaY < 0 ? 1.06 : 0.94;
        scale *= factor;
        scale = Math.max(0.05, Math.min(5, scale));
        reprojectAll(); draw();
    }, { passive: false });

    canvas.addEventListener('contextmenu', function(e) { e.preventDefault(); });
    canvas.addEventListener('dblclick', function() { setView(30, -30); });

    function updateAngleSliders() {
        pitchSlider.value = Math.round(angleX * 180 / Math.PI);
        yawSlider.value = Math.round(angleZ * 180 / Math.PI);
    }

    function onAngleSlider() {
        angleX = parseInt(pitchSlider.value) * Math.PI / 180;
        angleZ = parseInt(yawSlider.value) * Math.PI / 180;
        reprojectAll(); draw();
    }

    function setView(pitchDeg, yawDeg) {
        stopAutoRotate();
        angleX = pitchDeg * Math.PI / 180;
        angleZ = yawDeg * Math.PI / 180;
        updateAngleSliders();
        reprojectAll(); draw();
    }

    function autoRotate() {
        if (autoRotating) { stopAutoRotate(); return; }
        autoRotating = true;
        function step() {
            if (!autoRotating) return;
            angleZ += 0.008;
            updateAngleSliders();
            reprojectAll(); draw();
            autoRotateId = requestAnimationFrame(step);
        }
        autoRotateId = requestAnimationFrame(step);
    }

    function stopAutoRotate() {
        autoRotating = false;
        if (autoRotateId) cancelAnimationFrame(autoRotateId);
    }

            case 'ripple': customExpr = 'sin(x)*cos(y)'; break;
            case 'hyperboloid': customExpr = 'sqrt(1+x^2+y^2)'; break;
            case 'custom': break;
        }
        customEqInput.value = customExpr;
        funcOverlay.textContent = 'f(x,y) = ' + customExpr.replace(/\^2/g,'²').replace(/sqrt/g,'√').replace(/\*/g,'');
        updateBinaryFunc(); findGlobalExtrema(); reprojectAll(); draw();
